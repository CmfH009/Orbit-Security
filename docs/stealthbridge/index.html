Python Source Install & Safe-Harbor Usage

pip install "orbit-security @ git+https://github.com/CmfH009/Orbit-Security.git" 
from orbit_security.scanner import PublicHostOnlyRoE, create_guarded_httpx_client

# 1. Enforce read-only public-host Rules of Engagement (SSRF & private-IP guard)
roe = PublicHostOnlyRoE(
    allowed_hosts=["example.com"],
    requests_per_second=1.0,
)

# 2. Inspect public response headers & passive DNS/TLS metadata safely
with create_guarded_httpx_client(roe_engine=roe) as client:
    resp = client.get("https://example.com/")
    print(resp.status_code, resp.headers.get("strict-transport-security"))
