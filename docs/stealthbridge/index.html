from orbit_security.stealth_bridge_sdk import StealthSession, StealthBridgeConfig

# 1. Initialize configuration with authentic taskbar browser hook
config = StealthBridgeConfig(
    seat_license_key="STEALTH-PRO-XXXX",
    humanize_latency=True,
    base_typing_delay_ms=45.0,
    typing_jitter_ms=30.0,
    auto_solve_turnstile=True,
)

# 2. Attach to browser session with authentic cookies & TLS state
session = StealthSession(config=config)
session.connect()

# 3. Navigate securely with zero webdriver leaks
session.navigate("https://target-service.com/login")

# 4. Organic Bézier cursor movement and natural typing cadence
session.human_click(target_coords=(340.0, 480.0))
duration = session.human_type("hello@company.com")
print(f"Typed naturally in {duration}s")
